Un formulaire HTML ne sait envoyer que GET et POST

L'attribut method d'un formulaire n'accepte que GET et POST. Pour demander un PUT ou un DELETE, le formulaire envoie un POST accompagné d'un champ caché _method, écrit par la directive Blade @method('PUT'). Laravel lit ce champ et traite la requête comme un PUT : c'est l'usurpation de méthode (method spoofing).

HTML produit par le formulaire de modification (extrait simplifié) :

<form method="POST" action="http://localhost:8000/posts/3">
    <input type="hidden" name="_token" value="(jeton CSRF)">
    <input type="hidden" name="_method" value="PUT">
</form>

On n'édite jamais une migration déjà exécutée

php artisan migrate n'exécute que les migrations absentes de la table migrations. Modifier un fichier déjà exécuté n'a donc aucun effet sur votre base, et casse celle des personnes qui clonent le dépôt.

Unicité à la modification et scopes

  1. La règle unique:posts,slug refuse un slug dès qu'il existe dans la table, y compris quand c'est celui de l'article en cours. À la modification, on demande donc d'ignorer cet article : Rule::unique('posts', 'slug')->ignore($post).
  2. Un scope local donne un nom à une condition répétée : Post::published() remplace where('is_published', true) partout où on veut les articles publiés.

namespace App\Http\Controllers;

use App\Models\Post;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Validation\Rule;
use Illuminate\View\View;

class PostController extends Controller
{
    public function index(): View
    {
        return view('posts.index', [
            'heading' => 'Tous les articles',
            'posts' => Post::latest('id')->paginate(4),
        ]);
    }

    public function archive(string $year): View
    {
        return view('posts.index', [
            'heading' => 'Articles de '.$year,
            'posts' => Post::published()->whereYear('published_at', $year)->latest('id')->paginate(4),
        ]);
    }

    public function create(): View
    {
        return view('posts.create');
    }

    public function store(Request $request): RedirectResponse
    {
        $validated = $request->validate([
            'title' => ['required', 'string', 'max:255'],
            'slug' => ['required', 'string', 'max:255', 'regex:/^[a-z0-9-]+$/', 'unique:posts,slug'],
            'excerpt' => ['nullable', 'string', 'max:255'],
            'body' => ['required', 'string', 'min:20'],
            'is_published' => ['nullable', 'boolean'],
        ]);

        $validated['is_published'] = $request->boolean('is_published');
        $validated['published_at'] = $validated['is_published'] ? now() : null;

        $post = Post::create($validated);

        return redirect()->route('posts.show', $post);
    }

    public function show(Post $post): View
    {
        return view('posts.show', compact('post'));
    }

    public function edit(Post $post): View
    {
        return view('posts.edit', compact('post'));
    }

    public function update(Request $request, Post $post): RedirectResponse
    {
        $validated = $request->validate([
            'title' => ['required', 'string', 'max:255'],
            'slug' => ['required', 'string', 'max:255', 'regex:/^[a-z0-9-]+$/', Rule::unique('posts', 'slug')->ignore($post)],
            'excerpt' => ['nullable', 'string', 'max:255'],
            'body' => ['required', 'string', 'min:20'],
            'is_published' => ['nullable', 'boolean'],
        ]);

        $validated['is_published'] = $request->boolean('is_published');
        $validated['published_at'] = $validated['is_published'] ? ($post->published_at ?? now()) : null;

        $post->update($validated);

        return redirect()->route('posts.show', $post);
    }

    public function destroy(Post $post): RedirectResponse
    {
        $post->delete();

        return redirect()->route('posts.index');
    }
