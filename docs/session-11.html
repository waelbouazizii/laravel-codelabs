Deuxième passage sur MVC : qui est là, et qu'a-t-il le droit de faire ? 15 min

L'authentification répond à la question « qui êtes-vous ? ». Après Auth::attempt(), la session garde l'identifiant de l'utilisateur, et le navigateur renvoie le cookie de session avec chaque requête : Laravel retrouve ainsi l'utilisateur sans redemander le mot de passe.

L'autorisation répond à la question « avez-vous le droit ? ». Un Gate est une règle nommée (« gérer les catégories ») ; une Policy regroupe les règles qui portent sur un modèle (« modifier cet article »).

Idée reçue à écarter : « cacher le bouton suffit ». @can ne change que l'affichage ; une adresse tapée à la main arrive quand même au contrôleur, et c'est lui qui doit refuser.

À retenir

Pourquoi ? Le middleware décide si la requête peut entrer : il vérifie l'identité. La policy décide si cet utilisateur peut agir sur cet article : elle vérifie les droits. Les deux sont nécessaires, et dans cet ordre : sans identité, il n'y a personne à qui appliquer une règle.

<?php

namespace App\Http\Requests;

use Illuminate\Contracts\Validation\ValidationRule;
use Illuminate\Foundation\Http\FormRequest;

class RegisterRequest extends FormRequest
{
    /**
     * Determine if the user is authorized to make this request.
     */
    public function authorize(): bool
    {
        return true;
    }

    /**
     * Get the validation rules that apply to the request.
     *
     * @return array<string, ValidationRule|array<mixed>|string>
     */
    public function rules(): array
    {
        return [
            'name' => ['required', 'string', 'max:255'],
            'email' => ['required', 'string', 'email', 'max:255', 'unique:users,email'],
            'password' => ['required', 'string', 'min:8', 'confirmed'],

    }
}

La règle confirmed compare password avec le champ password_confirmation ; ce second champ n'a pas besoin de règle à lui. Comme en session 09, authorize() renvoie true : tout visiteur peut s'inscrire.

app/Http/Controllers/Auth/RegisterController.php (contenu complet, à la place du fichier généré) :

<?php

namespace App\Http\Controllers\Auth;

use App\Http\Controllers\Controller;
use App\Http\Requests\RegisterRequest;
use App\Models\User;
use Illuminate\Http\RedirectResponse;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Hash;
use Illuminate\View\View;

class RegisterController extends Controller
{
    public function create(): View
    {
        return view('auth.register');
    }

    public function store(RegisterRequest $request): RedirectResponse
    {
        $validated = $request->validated();

        $user = User::create([
            'name' => $validated['name'],
            'email' => $validated['email'],
            'password' => Hash::make($validated['password']),
        ]);

        Auth::login($user);

        $request->session()->regenerate();

        return redirect()->route('posts.index')->with('success', 'Compte créé : bienvenue, '.$user->name.'.');
    }
}

<?php

namespace App\Http\Controllers\Auth;

use App\Http\Controllers\Controller;
use App\Http\Requests\LoginRequest;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\View\View;

class LoginController extends Controller
{
    public function create(): View
    {
        return view('auth.login');
    }

    public function store(LoginRequest $request): RedirectResponse
    {
        if (! Auth::attempt($request->validated())) {
            return back()
                ->withErrors(['email' => 'Ces identifiants ne correspondent à aucun compte.'])
                ->onlyInput('email');
        }

        $request->session()->regenerate();

        return redirect()->intended(route('posts.index'))->with('success', 'Vous êtes connecté.');
    }

    public function destroy(Request $request): RedirectResponse
    {
        Auth::logout();

        $request->session()->invalidate();
        $request->session()->regenerateToken();

        return redirect()->route('home')->with('success', 'Vous êtes déconnecté.');
    }
}
  • attempt() reçoit seulement l'email et le mot de passe ; il cherche l'utilisateur par son email, puis compare avec Hash::check().
  • onlyInput('email') garde l'email dans old(), jamais le mot de passe.
  • intended() renvoie vers la page que le middleware auth a interrompue (Étape 4), ou vers /posts.

            'heading' => 'Catégorie : '.$category->name,
            'posts' => $category->posts()
                ->published()
                ->with('user', 'category', 'tags')
                ->latest('published_at')
                ->paginate(4),
        ]);
    }

    public function byTag(Tag $tag): View
    {
        return view('posts.index', [
            'heading' => 'Tag : '.$tag->name,
            'posts' => $tag->posts()
                ->published()
                ->with('user', 'category', 'tags')
                ->latest('published_at')
                ->paginate(4),
        ]);
    }

    public function create(): View
    {
        return view('posts.create', [
            'categories' => Category::orderBy('name')->get(),
            'tags' => Tag::orderBy('name')->get(),
        ]);
    }

    public function store(StorePostRequest $request): RedirectResponse
    {
        $validated = $request->validated();

        $validated['is_published'] = $request->boolean('is_published');
        $validated['published_at'] = $validated['is_published'] ? now() : null;

        $post = $request->user()->posts()->create(Arr::except($validated, ['tags']));
        $post->tags()->sync($validated['tags'] ?? []);

        return redirect()->route('posts.show', $post)->with('success', 'Article créé.');
    }

    public function show(Post $post): View
    {
        $post->load('user', 'category', 'tags');

        $related = $post->category
            ? $post->category->posts()
                ->published()
                ->where('id', '!=', $post->id)
                ->with('user', 'category', 'tags')
                ->latest('published_at')
                ->take(3)
                ->get()
            : collect();

        return view('posts.show', compact('post', 'related'));
    }

    public function edit(Post $post): View
    {
        $post->load('tags');

        return view('posts.edit', [
            'post' => $post,
            'categories' => Category::orderBy('name')->get(),
            'tags' => Tag::orderBy('name')->get(),
        ]);
    }

    public function update(UpdatePostRequest $request, Post $post): RedirectResponse
    {
        $validated = $request->validated();

        $validated['is_published'] = $request->boolean('is_published');
        $validated['published_at'] = $validated['is_published'] ? ($post->published_at ?? now()) : null;

        $post->update(Arr::except($validated, ['tags']));
        $post->tags()->sync($validated['tags'] ?? []);

        return redirect()->route('posts.show', $post)->with('success', 'Article mis à jour.');
    }

    public function destroy(Post $post): RedirectResponse
    {
        $post->delete();

        return redirect()->route('posts.index')->with('success', 'Article supprimé.');
    }

    public function togglePublication(Post $post): RedirectResponse
    {
        $isPublished = ! $post->is_published;

        $post->update([
            'is_published' => $isPublished,
            'published_at' => $isPublished ? now() : null,
        ]);

        return redirect()->route('posts.show', $post)->with('success', $isPublished ? 'Article publié.' : 'Article dépublié.');
    }

Dans routes/web.php, on place Route::resource('posts', PostController::class)->only(['index', 'show']) avant le groupe auth. Qu'affiche alors /posts/create ?

Laravel prend la première route qui correspond. posts/{post} accepte « create », la liaison de modèle ne trouve aucun article, et la fonction render de la session 09 répond.

Le formulaire d'article n'a aucun champ user_id. D'où vient l'auteur du nouvel article ?

La relation posts() du User remplit user_id avec l'id de l'utilisateur connecté ; aucun formulaire ne peut le choisir.

<?php

namespace App\Http\Controllers;

use App\Http\Requests\StorePostRequest;
use App\Http\Requests\UpdatePostRequest;
use App\Models\Category;
use App\Models\Post;
use App\Models\Tag;
use Illuminate\Http\RedirectResponse;
use Illuminate\Support\Arr;
use Illuminate\Support\Facades\Gate;
use Illuminate\View\View;

class PostController extends Controller
{
    public function index(): View
    {
        return view('posts.index', [
            'heading' => 'Tous les articles',
            'posts' => Post::with('user', 'category', 'tags')->latest('id')->paginate(4),
        ]);
    }

    public function archive(string $year): View
    {
        return view('posts.index', [
            'heading' => 'Articles de '.$year,
            'posts' => Post::published()->with('user', 'category', 'tags')->whereYear('published_at', $year)->latest('id')->paginate(4),
        ]);
    }

    public function byCategory(Category $category): View
    {
        return view('posts.index', [
            'heading' => 'Catégorie : '.$category->name,
            'posts' => $category->posts()
                ->published()
                ->with('user', 'category', 'tags')
                ->latest('published_at')
                ->paginate(4),
        ]);
    }

    public function byTag(Tag $tag): View
    {
        return view('posts.index', [
            'heading' => 'Tag : '.$tag->name,
            'posts' => $tag->posts()
                ->published()
                ->with('user', 'category', 'tags')
                ->latest('published_at')
                ->paginate(4),
        ]);
    }

    public function create(): View
    {
        return view('posts.create', [
            'categories' => Category::orderBy('name')->get(),
            'tags' => Tag::orderBy('name')->get(),
        ]);
    }

    public function store(StorePostRequest $request): RedirectResponse
    {
        $validated = $request->validated();

        $validated['is_published'] = $request->boolean('is_published');
        $validated['published_at'] = $validated['is_published'] ? now() : null;

        $post = $request->user()->posts()->create(Arr::except($validated, ['tags']));
        $post->tags()->sync($validated['tags'] ?? []);

        return redirect()->route('posts.show', $post)->with('success', 'Article créé.');
    }

    public function show(Post $post): View
    {
        $post->load('user', 'category', 'tags');

        $related = $post->category
            ? $post->category->posts()
                ->published()
                ->where('id', '!=', $post->id)
                ->with('user', 'category', 'tags')
                ->latest('published_at')
                ->take(3)
                ->get()
            : collect();

        return view('posts.show', compact('post', 'related'));
    }

    public function edit(Post $post): View
    {
        Gate::authorize('update', $post);

        $post->load('tags');

        return view('posts.edit', [
            'post' => $post,
            'categories' => Category::orderBy('name')->get(),
            'tags' => Tag::orderBy('name')->get(),
        ]);
    }

    public function update(UpdatePostRequest $request, Post $post): RedirectResponse
    {
        $validated = $request->validated();

        $validated['is_published'] = $request->boolean('is_published');
        $validated['published_at'] = $validated['is_published'] ? ($post->published_at ?? now()) : null;

        $post->update(Arr::except($validated, ['tags']));
        $post->tags()->sync($validated['tags'] ?? []);

        return redirect()->route('posts.show', $post)->with('success', 'Article mis à jour.');
    }

    public function destroy(Post $post): RedirectResponse
    {
        Gate::authorize('delete', $post);

        $post->delete();

        return redirect()->route('posts.index')->with('success', 'Article supprimé.');
    }

    public function togglePublication(Post $post): RedirectResponse
    {
        Gate::authorize('update', $post);

        $isPublished = ! $post->is_published;

        $post->update([
            'is_published' => $isPublished,
            'published_at' => $isPublished ? now() : null,
        ]);

        return redirect()->route('posts.show', $post)->with('success', $isPublished ? 'Article publié.' : 'Article dépublié.');
    }
}
