Deuxième passage sur MVC : ce qui se passe avant le contrôleur 15 min

Jusqu'ici le contrôleur faisait tout : vérifier, enregistrer, répondre. Cette session place deux filtres devant lui : le middleware décide si la requête peut entrer, la Form Request décide si ses données sont valides.

Idée reçue à écarter : « le contrôleur appelle la Form Request ». C'est faux : Laravel la crée et la valide avant d'entrer dans store() ; si elle échoue, aucune ligne de store() ne s'exécute.

À retenir

Pourquoi ? Une requête d'écriture traverse dans l'ordre le middleware, puis la Form Request (authorize(), puis rules()), puis le contrôleur. Chaque filtre peut répondre seul (503, 403, ou 302 avec les erreurs) : le contrôleur ne voit que des requêtes autorisées et des données valides, et il ajoute le message flash avant de rediriger.

Erreurs fréquentes : « Le contrôleur valide encore » : une ligne $request->validate() restée dans store() fait valider deux fois les mêmes données. Une fois la Form Request en place, le contrôleur ne garde que $request->validated().

<?php

namespace App\Http\Requests;

use Illuminate\Contracts\Validation\ValidationRule;
use Illuminate\Foundation\Http\FormRequest;

class StorePostRequest extends FormRequest
{
    /**
     * Determine if the user is authorized to make this request.
     */
    public function authorize(): bool
    {
        return true;
    }

    /**
     * Get the validation rules that apply to the request.
     *
     * @return array<string, ValidationRule|array<mixed>|string>
     */
    public function rules(): array
    {
        return [
            'title' => ['required', 'string', 'max:255'],
            'slug' => ['required', 'string', 'max:255', 'regex:/^[a-z0-9-]+$/', 'unique:posts,slug'],
            'excerpt' => ['nullable', 'string', 'max:255'],
            'body' => ['required', 'string', 'min:20'],
            'is_published' => ['nullable', 'boolean'],
            'category_id' => ['nullable', 'integer', 'exists:categories,id'],
            'tags' => ['nullable', 'array'],
            'tags.*' => ['integer', 'exists:tags,id'],
        ];
    }
}

<?php

namespace App\Http\Controllers;

use App\Http\Requests\StorePostRequest;
use App\Models\Category;
use App\Models\Post;
use App\Models\Tag;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Arr;
use Illuminate\Validation\Rule;
use Illuminate\View\View;

class PostController extends Controller
{
    public function index(): View
    {
        return view('posts.index', [
            'heading' => 'Tous les articles',
            'posts' => Post::with('user', 'category', 'tags')->latest('id')->paginate(4),
        ]);
    }

    public function archive(string $year): View
    {
        return view('posts.index', [
            'heading' => 'Articles de '.$year,
            'posts' => Post::published()->with('user', 'category', 'tags')->whereYear('published_at', $year)->latest('id')->paginate(4),
        ]);
    }

    public function byCategory(Category $category): View
    {
        return view('posts.index', [
            'heading' => 'Catégorie : '.$category->name,
            'posts' => $category->posts()
                ->published()
                ->with('user', 'category', 'tags')
                ->latest('published_at')
                ->paginate(4),
        ]);
    }

    public function byTag(Tag $tag): View
    {
        return view('posts.index', [
            'heading' => 'Tag : '.$tag->name,
            'posts' => $tag->posts()
                ->published()
                ->with('user', 'category', 'tags')
                ->latest('published_at')
                ->paginate(4),
        ]);
    }

    public function create(): View
    {
        return view('posts.create', [
            'categories' => Category::orderBy('name')->get(),
            'tags' => Tag::orderBy('name')->get(),
        ]);
    }

    public function store(StorePostRequest $request): RedirectResponse
    {
        $validated = $request->validated();

        $validated['is_published'] = $request->boolean('is_published');
        $validated['published_at'] = $validated['is_published'] ? now() : null;

        $post = Post::create(Arr::except($validated, ['tags']));
        $post->tags()->sync($validated['tags'] ?? []);

        return redirect()->route('posts.show', $post);
    }

    public function show(Post $post): View
    {
        $post->load('user', 'category', 'tags');

        $related = $post->category
            ? $post->category->posts()
                ->published()
                ->where('id', '!=', $post->id)
                ->with('user', 'category', 'tags')
                ->latest('published_at')
                ->take(3)
                ->get()
            : collect();

        return view('posts.show', compact('post', 'related'));
    }

    public function edit(Post $post): View
    {
        $post->load('tags');

        return view('posts.edit', [
            'post' => $post,
            'categories' => Category::orderBy('name')->get(),
            'tags' => Tag::orderBy('name')->get(),
        ]);
    }

    public function update(Request $request, Post $post): RedirectResponse
    {
        $validated = $request->validate([
            'title' => ['required', 'string', 'max:255'],
            'slug' => ['required', 'string', 'max:255', 'regex:/^[a-z0-9-]+$/', Rule::unique('posts', 'slug')->ignore($post)],
            'excerpt' => ['nullable', 'string', 'max:255'],
            'body' => ['required', 'string', 'min:20'],
            'is_published' => ['nullable', 'boolean'],
            'category_id' => ['nullable', 'integer', 'exists:categories,id'],
            'tags' => ['nullable', 'array'],
            'tags.*' => ['integer', 'exists:tags,id'],
        ]);

        $validated['is_published'] = $request->boolean('is_published');
        $validated['published_at'] = $validated['is_published'] ? ($post->published_at ?? now()) : null;

        $post->update(Arr::except($validated, ['tags']));
        $post->tags()->sync($validated['tags'] ?? []);

        return redirect()->route('posts.show', $post);
    }

    public function destroy(Post $post): RedirectResponse
    {
        $post->delete();

        return redirect()->route('posts.index');
    }

    public function togglePublication(Post $post): RedirectResponse
    {
        $isPublished = ! $post->is_published;

        $post->update([
            'is_published' => $isPublished,
            'published_at' => $isPublished ? now() : null,
        ]);

        return redirect()->route('posts.show', $post);
    }
}

<?php

namespace App\Http\Controllers;

use App\Http\Requests\StorePostRequest;
use App\Http\Requests\UpdatePostRequest;
use App\Models\Category;
use App\Models\Post;
use App\Models\Tag;
use Illuminate\Http\RedirectResponse;
use Illuminate\Support\Arr;
use Illuminate\View\View;

class PostController extends Controller
{
    public function index(): View
    {
        return view('posts.index', [
            'heading' => 'Tous les articles',
            'posts' => Post::with('user', 'category', 'tags')->latest('id')->paginate(4),
        ]);
    }

    public function archive(string $year): View
    {
        return view('posts.index', [
            'heading' => 'Articles de '.$year,
            'posts' => Post::published()->with('user', 'category', 'tags')->whereYear('published_at', $year)->latest('id')->paginate(4),
        ]);
    }

    public function byCategory(Category $category): View
    {
        return view('posts.index', [
            'heading' => 'Catégorie : '.$category->name,
            'posts' => $category->posts()
                ->published()
                ->with('user', 'category', 'tags')
                ->latest('published_at')
                ->paginate(4),
        ]);
    }

    public function byTag(Tag $tag): View
    {
        return view('posts.index', [
            'heading' => 'Tag : '.$tag->name,
            'posts' => $tag->posts()
                ->published()
                ->with('user', 'category', 'tags')
                ->latest('published_at')
                ->paginate(4),
        ]);
    }

    public function create(): View
    {
        return view('posts.create', [
            'categories' => Category::orderBy('name')->get(),
            'tags' => Tag::orderBy('name')->get(),
        ]);
    }

    public function store(StorePostRequest $request): RedirectResponse
    {
        $validated = $request->validated();

        $validated['is_published'] = $request->boolean('is_published');
        $validated['published_at'] = $validated['is_published'] ? now() : null;

        $post = Post::create(Arr::except($validated, ['tags']));
        $post->tags()->sync($validated['tags'] ?? []);

        return redirect()->route('posts.show', $post)->with('success', 'Article créé.');
    }

    public function show(Post $post): View
    {
        $post->load('user', 'category', 'tags');

        $related = $post->category
            ? $post->category->posts()
                ->published()
                ->where('id', '!=', $post->id)
                ->with('user', 'category', 'tags')
                ->latest('published_at')
                ->take(3)
                ->get()
            : collect();

        return view('posts.show', compact('post', 'related'));
    }

    public function edit(Post $post): View
    {
        $post->load('tags');

        return view('posts.edit', [
            'post' => $post,
            'categories' => Category::orderBy('name')->get(),
            'tags' => Tag::orderBy('name')->get(),
        ]);
    }

    public function update(UpdatePostRequest $request, Post $post): RedirectResponse
    {
        $validated = $request->validated();

        $validated['is_published'] = $request->boolean('is_published');
        $validated['published_at'] = $validated['is_published'] ? ($post->published_at ?? now()) : null;

        $post->update(Arr::except($validated, ['tags']));
        $post->tags()->sync($validated['tags'] ?? []);

        return redirect()->route('posts.show', $post)->with('success', 'Article mis à jour.');
    }

    public function destroy(Post $post): RedirectResponse
    {
        $post->delete();

        return redirect()->route('posts.index')->with('success', 'Article supprimé.');
    }

    public function togglePublication(Post $post): RedirectResponse
    {
        $isPublished = ! $post->is_published;

        $post->update([
            'is_published' => $isPublished,
            'published_at' => $isPublished ? now() : null,
        ]);

        return redirect()->route('posts.show', $post)->with('success', $isPublished ? 'Article publié.' : 'Article dépublié.');
    }
}

<?php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class EnsureSiteIsWritable
{
    /**
     * Handle an incoming request.
     *
     * @param  Closure(Request): (Response)  $next
     */
    public function handle(Request $request, Closure $next): Response
    {
        if (config('site.read_only') && ! $request->isMethodSafe()) {
            abort(503, 'Le site est en lecture seule.');
        }

        return $next($request);
    }
}

isMethodSafe() renvoie true pour les méthodes qui ne modifient rien : GET et HEAD (ainsi que OPTIONS et TRACE). Les formulaires qui utilisent @method('PUT'), @method('PATCH') ou @method('DELETE') sont bien vus comme PUT, PATCH ou DELETE, donc bloqués.
